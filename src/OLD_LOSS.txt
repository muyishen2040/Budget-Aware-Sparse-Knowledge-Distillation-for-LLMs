def compute_compression_kd_loss(compressedk_probs, compressed_indices, student_logits, topk_teacher_probs, topk_indices, labels, temperature=1.0, alpha=0.5):
    shift_logits = student_logits[..., :-1, :].contiguous().float()
    shift_labels = labels[..., 1:].contiguous()
    
    ce_loss = F.cross_entropy(shift_logits.view(-1, shift_logits.size(-1)), shift_labels.view(-1))
    
    shift_compressed_teacher_probs = compressedk_probs[..., :-1, :].contiguous().float()
    shift_compressed_indices = compressed_indices[..., :-1, :].contiguous()
    
    # Compute full log_softmax first to penalize non-topk probability mass
    student_full_log_probs = F.log_softmax(shift_logits / temperature, dim=-1)
    
    # Gather student log_probs based on teacher's compressed_indices
    student_log_probs = torch.gather(student_full_log_probs, dim=-1, index=shift_compressed_indices)
    
    # Renormalize teacher probabilities over the compressed support
    teacher_probs = shift_compressed_teacher_probs / shift_compressed_teacher_probs.sum(dim=-1, keepdim=True)
    
    k = shift_compressed_indices.size(-1)
    kl = F.kl_div(
        student_log_probs.view(-1, k),
        teacher_probs.view(-1, k),
        reduction='none'
    )
    kl = kl.sum(dim=-1).view(*shift_labels.shape)
    valid_mask = (shift_labels != -100)
    
    if valid_mask.any():
        kl_loss = kl[valid_mask].mean() * (temperature ** 2)
    else:
        kl_loss = torch.zeros((), device=shift_logits.device, dtype=shift_logits.dtype)
    
    loss = alpha * ce_loss + (1 - alpha) * kl_loss
    
    return loss, ce_loss, kl_loss
